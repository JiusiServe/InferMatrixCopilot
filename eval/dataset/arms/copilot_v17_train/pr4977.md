```suggestion
_hub_locks: dict[str, threading.Lock] = {}
_hub_locks_guard = threading.Lock()


def _get_hub_module(repo_id: str):
    with _hub_locks_guard:
        lock = _hub_locks.setdefault(repo_id, threading.Lock())
    with lock:
        if repo_id not in _hub_modules:
            _hub_modules[repo_id] = _load_hub_module(repo_id)
        return _hub_modules[repo_id]
```
